cmake_minimum_required(VERSION 3.20)
project(LaserVaultHeist VERSION 1.0.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(GLFW_BUILD_DOCS OFF CACHE BOOL "" FORCE)
set(GLFW_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(GLFW_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(GLFW_INSTALL OFF CACHE BOOL "" FORCE)
add_subdirectory(third_party/glfw EXCLUDE_FROM_ALL)
add_library(glad STATIC third_party/glad/src/gl.c)
target_include_directories(glad PUBLIC third_party/glad/include)
add_library(imgui STATIC third_party/imgui/imgui.cpp third_party/imgui/imgui_draw.cpp
 third_party/imgui/imgui_tables.cpp third_party/imgui/imgui_widgets.cpp
 third_party/imgui/backends/imgui_impl_glfw.cpp third_party/imgui/backends/imgui_impl_opengl3.cpp)
target_include_directories(imgui PUBLIC third_party/imgui third_party/imgui/backends)
target_link_libraries(imgui PUBLIC glfw)
add_executable(LaserVaultHeist src/main.cpp src/animation.cpp src/graphics.cpp src/scene.cpp src/game.cpp src/ui.cpp src/renderer.cpp src/world.cpp src/recorder.cpp src/director.cpp src/route.cpp)
target_include_directories(LaserVaultHeist PRIVATE include third_party/glm)
target_include_directories(LaserVaultHeist SYSTEM PRIVATE third_party/stb)
target_link_libraries(LaserVaultHeist PRIVATE glad glfw imgui)
target_compile_definitions(LaserVaultHeist PRIVATE GLFW_INCLUDE_NONE)
if(MSVC)
 target_compile_options(LaserVaultHeist PRIVATE /W4 /permissive-)
else()
 target_compile_options(LaserVaultHeist PRIVATE -Wall -Wextra -Wpedantic)
 if(WIN32)
  target_link_options(LaserVaultHeist PRIVATE -static)
 endif()
endif()
add_custom_command(TARGET LaserVaultHeist POST_BUILD
 COMMAND ${CMAKE_COMMAND} -E copy_directory ${CMAKE_SOURCE_DIR}/shaders $<TARGET_FILE_DIR:LaserVaultHeist>/shaders
 COMMAND ${CMAKE_COMMAND} -E copy_directory ${CMAKE_SOURCE_DIR}/assets $<TARGET_FILE_DIR:LaserVaultHeist>/assets)
include(CTest)
if(BUILD_TESTING)
 add_executable(lvh_logic_tests tests/logic_tests.cpp src/game.cpp)
 target_include_directories(lvh_logic_tests PRIVATE include third_party/glm third_party/glad/include third_party/glfw/include)
 add_test(NAME gameplay_math COMMAND lvh_logic_tests)
endif()
install(TARGETS LaserVaultHeist RUNTIME DESTINATION .)
install(DIRECTORY assets shaders DESTINATION .)


if(BUILD_TESTING)
 add_executable(lvh_route_tests tests/route_tests.cpp src/game.cpp src/route.cpp)
 target_include_directories(lvh_route_tests PRIVATE include third_party/glm)
 add_test(NAME complete_gameplay_route COMMAND lvh_route_tests)
endif()


if(BUILD_TESTING)
 add_executable(lvh_director_tests tests/director_tests.cpp src/game.cpp src/route.cpp src/director.cpp)
 target_include_directories(lvh_director_tests PRIVATE include third_party/glm)
 add_test(NAME demo_and_slow_frames COMMAND lvh_director_tests)
endif()
